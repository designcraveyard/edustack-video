# Book Mode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bundle layout-gen's book-page generation as a sibling artifact to the existing edustack-video pipeline. Reuses brief / script / characters; produces curated A4-Portrait or A3-Landscape RGBA PNGs at 300 DPI via fal.ai (gpt-image-2 + birefnet fallback) with Gemini visual QA via openrouter.

**Architecture:** Brief-collector grows a *Book* step (page count 0–12, template shortlist, voice). After Phase 1 (script) gate, if `brief.book.page_count_target > 0`, orchestrator runs three new phases serially after video: **B1 book-plan** (curated page list, template + book-voice copy per page) → Gate B1 → **B2 book-render** (gpt-image-2, birefnet fallback for transparent BG, vision QA loop) → **B3 book-print-prep** (Pillow upscale + template-aware composite onto A4P/A3L transparent canvas at 300 DPI).

**Tech Stack:** Python (Pillow, fal-client, pydantic), Node stdlib HTTP (brief server). All external calls routed through fal.ai. No ImageMagick, no PDF tooling.

**Spec:** [docs/superpowers/specs/2026-05-17-book-mode-design.md](../specs/2026-05-17-book-mode-design.md)

---

## Open-question resolutions (made by the planner)

The spec leaves three open questions. This plan resolves them with the following assumptions; if any prove wrong at runtime, the impacted task includes a fallback.

1. **gpt-image-2 reference-image limits:** Assume up to 4 `image_urls` per call are reliable (matches OpenAI's documented gpt-image-1 limit, conservative for the fal proxy). v1 caps refs at 4 = 1 keyframe + 3 character images. If a page references more characters than that, the first 3 by `book_voice_copy` mention order are kept; the rest are described in the prompt's character brief paragraph instead of attached.
2. **Static-image QA fal route:** Confirmed by reading `scripts/lib/visual_analyzer.py`. Existing project uses `fal-ai/any-llm/vision` for keyframe + character-sheet QA. We extend `VisualAnalyzer` with `analyse_book_page()` using the same path and the existing `FalClient.any_llm_vision()` method.
3. **Pillow dependency status:** Confirmed in `requirements.txt` — `Pillow>=10.0.0` already pinned. No requirements change needed for v1.

- [ ] **Step 4.4: Wire JS in `form.js`**

At the top of `form.js`, after existing imports/initialization, add:

```js
// ---------- Book step ----------
const bookCountEl = document.getElementById("book-page-count");
const bookExtrasEl = document.getElementById("book-extras");
const galleryEl = document.getElementById("book-templates-gallery");
const hintEl = document.getElementById("book-templates-hint");
const selectedTemplates = new Set();

function toggleBookExtras() {
  const count = parseInt(bookCountEl.value || "0", 10);
  bookExtrasEl.hidden = count <= 0;
}

async function loadTemplateGallery() {
  try {
    const res = await fetch("/static/templates/manifest.json");
    if (!res.ok) return;
    const data = await res.json();
    galleryEl.replaceChildren();
    for (const t of data.templates) {
      const card = document.createElement("div");
      card.className = "card";
      card.dataset.id = t.id;

      const wf = document.createElement("img");
      wf.src = `/static/templates/wireframes/${t.wireframe}`;
      wf.alt = `${t.name} wireframe`;
      card.appendChild(wf);

      const name = document.createElement("div");
      name.className = "name";
      name.textContent = t.name;
      card.appendChild(name);

      const meta = document.createElement("div");
      meta.className = "meta";
      meta.textContent = `${t.aspect} • ${t.canvas} • ~${t.frequency}% • ${t.bestFor.join(", ")}`;
      card.appendChild(meta);

      const thumbs = document.createElement("div");
      thumbs.className = "thumbs";
      for (const r of t.refs.slice(0, 2)) {
        const img = document.createElement("img");
        img.src = `/static/templates/refs/${r}`;
        img.alt = "";
        thumbs.appendChild(img);
      }
      card.appendChild(thumbs);

      card.addEventListener("click", () => {
        if (selectedTemplates.has(t.id)) {
          selectedTemplates.delete(t.id);
          card.classList.remove("selected");
        } else {
          if (selectedTemplates.size >= 4) return;
          selectedTemplates.add(t.id);
          card.classList.add("selected");
        }
        updateHint();
      });

      galleryEl.appendChild(card);
    }
  } catch (e) {
    console.error("template gallery load failed", e);
  }
}

function updateHint() {
  const n = selectedTemplates.size;
  hintEl.textContent = n < 2
    ? `Pick at least 2 and at most 4 templates (${n} selected).`
    : n > 4
    ? `Too many. Maximum 4. (${n} selected)`
    : `${n} selected — ready.`;
}

bookCountEl.addEventListener("input", toggleBookExtras);
toggleBookExtras();
loadTemplateGallery();
```

```python
"""Template-aware composition rules for Phase B3 (book-print-prep).

Each rule answers: given the gpt-image-2 output (RGBA, native dims) and the
target A4P (2480x3508) or A3L (4961x3508) transparent canvas, where does the
illustration go and how is it scaled?

A rule is a dict:
  {
    "canvas":   "A4P" | "A3L",
    "scale":    float        # fraction of canvas long side the illustration occupies
    "position": "full" | "left" | "right" | "center" | "lower-center" | "upper-center"
    "margin":   int          # pixels of padding inside canvas, applied to all sides
  }

If a template ever changes, update the rule here and the change ripples to
every page using that template.
"""
from __future__ import annotations
from dataclasses import dataclass
from PIL import Image
from PIL.ImageFilter import UnsharpMask
from PIL.ImageOps import autocontrast


CANVAS_DIMS = {
    "A4P": (2480, 3508),   # 210×297mm @ 300 DPI
    "A3L": (4961, 3508),   # 420×297mm @ 300 DPI
}


@dataclass(frozen=True)
class ComposeRule:
    canvas: str
    scale: float
    position: str
    margin: int = 80


RULES: dict[str, ComposeRule] = {
    "full-bleed-with-text-zone":  ComposeRule("A3L", 1.00, "full",          margin=0),
    "vignette-on-page":           ComposeRule("A4P", 0.60, "lower-center",  margin=120),
    "split-layout":               ComposeRule("A3L", 0.60, "left",          margin=120),
    "scattered-spots":            ComposeRule("A4P", 0.90, "center",        margin=120),
    "full-spread-no-text":        ComposeRule("A3L", 1.00, "full",          margin=0),
    "illustrated-border":         ComposeRule("A4P", 0.95, "center",        margin=120),
    "character-text-pocket":      ComposeRule("A4P", 0.70, "upper-center",  margin=120),
    "connected-infographic":      ComposeRule("A4P", 0.95, "center",        margin=120),
    "spread-scene-plus-spots":    ComposeRule("A3L", 1.00, "full",          margin=0),
}


def _position_xy(canvas_w: int, canvas_h: int, art_w: int, art_h: int,
                 position: str, margin: int) -> tuple[int, int]:
    if position == "full":
        return (0, 0)
    if position == "left":
        return (margin, (canvas_h - art_h) // 2)
    if position == "right":
        return (canvas_w - art_w - margin, (canvas_h - art_h) // 2)
    if position == "center":
        return ((canvas_w - art_w) // 2, (canvas_h - art_h) // 2)
    if position == "lower-center":
        return ((canvas_w - art_w) // 2, canvas_h - art_h - margin)
    if position == "upper-center":
        return ((canvas_w - art_w) // 2, margin)
    # fallback
    return ((canvas_w - art_w) // 2, (canvas_h - art_h) // 2)


def compose(image: Image.Image, template: str) -> Image.Image:
    """Compose an RGBA illustration onto a transparent A4P or A3L canvas
    at the position prescribed by the template's rule. Returns a new RGBA image."""
    rule = RULES.get(template)
    if rule is None:
        raise KeyError(f"unknown book template: {template}")

    canvas_w, canvas_h = CANVAS_DIMS[rule.canvas]

    # mild auto-level + unsharp on RGB channels (preserve alpha)
    if image.mode != "RGBA":
        image = image.convert("RGBA")
    r, g, b, a = image.split()
    rgb = Image.merge("RGB", (r, g, b))
    rgb = autocontrast(rgb, cutoff=1)
    rgb = rgb.filter(UnsharpMask(radius=1.5, percent=70, threshold=2))
    r2, g2, b2 = rgb.split()
    image = Image.merge("RGBA", (r2, g2, b2, a))

    # scale: long-side of art fits scale × long-side of canvas
    target_long = int(max(canvas_w, canvas_h) * rule.scale) - 2 * rule.margin
    if target_long <= 0:
        target_long = max(canvas_w, canvas_h) - 2 * rule.margin
    src_long = max(image.width, image.height)
    factor = target_long / src_long
    art_w = max(1, int(image.width * factor))
    art_h = max(1, int(image.height * factor))
    art = image.resize((art_w, art_h), Image.LANCZOS)

    canvas = Image.new("RGBA", (canvas_w, canvas_h), (0, 0, 0, 0))
    x, y = _position_xy(canvas_w, canvas_h, art_w, art_h, rule.position, rule.margin)
    canvas.alpha_composite(art, (x, y))
    return canvas


def canvas_for(template: str) -> str:
    """Convenience for callers that need the canvas label (A4P|A3L)."""
    return RULES[template].canvas
```

CONTENT_TYPE_PREFS = {
    "narrative":      ["full-bleed-with-text-zone", "split-layout", "illustrated-border"],
    "educational":    ["scattered-spots", "connected-infographic", "split-layout"],
    "character_intro":["vignette-on-page", "character-text-pocket", "split-layout"],

def classify_scene(scene: dict) -> str:
    """Lightweight content classification from script scene metadata."""
    tags = " ".join([str(scene.get("intent", "")), str(scene.get("type", "")),
                     str(scene.get("description", ""))]).lower()
    if any(k in tags for k in ("intro", "introduce", "meet")):
        return "character_intro"
    if any(k in tags for k in ("dramatic", "climax", "reveal")):
        return "dramatic"
    if any(k in tags for k in ("explain", "concept", "how", "why", "step")):
        return "educational"
    return "narrative"


def pick_template(content_type: str, allowed: list[str]) -> str:
    for t in CONTENT_TYPE_PREFS.get(content_type, []):
        if t in allowed:
            return t
    return allowed[0]


def book_voice_rewrite(line: str, voice: str) -> str:
    """Deterministic fallback rewrite. The skill's LLM call typically supplies
    a better one; this is the safety net."""
    s = (line or "").strip()
    if not s:
        return ""
    if voice == "factual_calm":
        # strip exclamation, prefer present tense
        return s.rstrip("!.")
    if voice == "playful_rhyming":
        return s.rstrip(".") + "."
    # storybook_narrator default
    return s

```python
"""Phase B2 — book-render.

For each entry in book/plan.json, call fal-ai/gpt-image-2 with the template's
prompt + scene_prompt + reference images. If the output isn't transparent,
route through fal-ai/birefnet/v2. Then run vision QA via fal-ai/any-llm/vision
(Gemini 2.5 Pro). Up to 2 retries with corrective addendum per page.

Bounded: failures don't block the rest of the book — they're reported in
qa.json with status='failed' so the user can regen specific pages via the
/create-book-regen command.
"""
from __future__ import annotations
import argparse
import io
import json
import re
import sys
import urllib.request
from pathlib import Path

from PIL import Image

from scripts.lib.config import load as load_config
from scripts.lib.run_state import RunState
from scripts.lib.paths import RunPaths
from scripts.lib.fal_client import FalClient
from scripts.lib.visual_analyzer import VisualAnalyzer
from scripts.lib.vps_logger import VPSLogger


TRANSPARENT_DIRECTIVE = (
    "\n\nTransparent background. Illustration only. The {text_zone_hint} region must "
    "be left fully transparent — no painted background, no parchment, no light wash. "
    "No text, no letters, no numbers anywhere in the image."
)


def load_templates_md(plugin_root: Path) -> dict[str, str]:
    """Parse skills/book-plan/references/templates.md into {template_id: prompt_pattern}."""
    md = (plugin_root / "skills" / "book-plan" / "references" / "templates.md").read_text(encoding="utf-8")
    sections = re.split(r"(?m)^## \d+\.\s*([\w-]+)\s*$", md)
    out: dict[str, str] = {}
    for i in range(1, len(sections), 2):
        name = sections[i].strip()
        body = sections[i + 1]
        m = re.search(r"```(?:text|md)?\n(.+?)\n```", body, re.S)
        if m:
            out[name] = m.group(1).strip()
    return out


def render_prompt(pattern: str, scene_prompt: str, prompt_params: dict, text_zone_hint: str) -> str:
    """Substitute {placeholders} in a template pattern and append the transparent-BG directive."""
    p = pattern
    p = p.replace("{scene_description}", scene_prompt)
    for k, v in (prompt_params or {}).items():
        p = p.replace("{" + k + "}", str(v))
    # Any remaining {placeholders} get a benign default
    p = re.sub(r"\{[^}]+\}", "neutral", p)
    return p + TRANSPARENT_DIRECTIVE.format(text_zone_hint=text_zone_hint or "text zone")


def _download(url: str) -> bytes:
    with urllib.request.urlopen(url, timeout=120) as r:
        return r.read()


def _is_transparent_enough(img: Image.Image) -> bool:
    """Border-alpha heuristic: <2% of border pixels with alpha>5 → ok."""
    if img.mode != "RGBA":
        return False
    w, h = img.size
    alpha = img.split()[-1]
    px = alpha.load()
    samples = []
    for x in range(0, w, max(1, w // 50)):
        samples.append(px[x, 0]); samples.append(px[x, h - 1])
    for y in range(0, h, max(1, h // 50)):
        samples.append(px[0, y]); samples.append(px[w - 1, y])
    if not samples:
        return False
    opaque = sum(1 for v in samples if v > 5)
    return (opaque / len(samples)) < 0.02


def _to_data_url(path: Path) -> str:
    import base64, mimetypes
    mime, _ = mimetypes.guess_type(str(path))
    mime = mime or "image/png"
    data = base64.b64encode(path.read_bytes()).decode("ascii")
    return f"data:{mime};base64,{data}"


def render_one_page(
    page: dict, run_dir: Path, fal: FalClient, analyzer: VisualAnalyzer,
    templates_md: dict[str, str], character_brief: str, max_retries: int = 2,
) -> dict:
    page_no = int(page["page_no"])
    tmpl = page["template"]
    canvas = page["canvas"]
    size = "1024x1456" if canvas == "A4P" else "1456x1024"
    pattern = templates_md.get(tmpl, "Generate a children's book illustration: {scene_description}. {art_style_description}.")
    prompt = render_prompt(pattern, page["scene_prompt"], page.get("prompt_params") or {}, page.get("text_zone_hint") or "")

    # Build image refs — keyframe first, then up to 3 character refs (cap 4)
    image_urls: list[str] = []
    kf = page.get("keyframe_ref")
    if kf:
        kf_path = run_dir / kf
        if kf_path.exists():
            image_urls.append(_to_data_url(kf_path))
    for cref in (page.get("character_refs") or [])[:3]:
        cref_path = run_dir / cref
        if cref_path.exists():
            image_urls.append(_to_data_url(cref_path))
    image_urls = image_urls[:4]

    out_dir = run_dir / "book" / "pages"
    out_dir.mkdir(parents=True, exist_ok=True)
    out_png = out_dir / f"page-{page_no:02d}.png"
    qa_json = out_dir / f"page-{page_no:02d}.qa.json"

    attempt = 0
    last_qa: dict = {}
    addendum = ""
    while attempt <= max_retries:
        result = fal.gpt_image_2_book_page(
            prompt=(prompt + (("\n\n" + addendum) if addendum else "")),
            image_urls=image_urls or None,
            size=size,
        )
        # Extract image URL — fal responses normally carry {"images":[{"url":...}]}
        try:
            url = result["images"][0]["url"]
        except Exception:
            raise RuntimeError(f"gpt-image-2 returned unexpected payload for page {page_no}: {result!r}")

        raw = _download(url)
        img = Image.open(io.BytesIO(raw))
        if img.mode != "RGBA":
            img = img.convert("RGBA")

        if not _is_transparent_enough(img):
            # birefnet fallback
            try:
                br = fal.birefnet_remove_bg(image_url=url)
                br_url = br["images"][0]["url"] if "images" in br else br["image"]["url"]
                img = Image.open(io.BytesIO(_download(br_url))).convert("RGBA")
            except Exception as e:  # noqa: BLE001
                # leave img as-is — QA will likely flag it
                print(f"[book-render] page {page_no}: birefnet fallback failed: {e}")

        img.save(out_png, format="PNG")

        last_qa = analyzer.analyse_book_page(
            image_path=out_png, prompt_used=prompt, character_brief=character_brief,
            page_no=page_no, scene_prompt=page["scene_prompt"],
        )
        if last_qa.get("verdict") == "APPROVED" or last_qa.get("_degraded"):
            break
        addendum = last_qa.get("corrective_addendum") or "Strictly transparent background and no text in the image."
        attempt += 1

    last_qa["attempts"] = attempt + 1
    last_qa["status"] = "ok" if last_qa.get("verdict") == "APPROVED" else "failed"
    qa_json.write_text(json.dumps(last_qa, indent=2), encoding="utf-8")
    return last_qa


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--run-dir", required=True)
    ap.add_argument("--only-page", type=int, default=0, help="If >0, render only that page_no.")
    args = ap.parse_args()
    run_dir = Path(args.run_dir).resolve()

    cfg = load_config(run_dir)
    rs = RunState.load_or_init(run_dir)
    rp = RunPaths(run_dir)

    plan_path = run_dir / "book" / "plan.json"
    if not plan_path.exists():
        print("[book-render] book/plan.json missing — run phase_book_plan first.", file=sys.stderr)
        return 2

    plan = json.loads(plan_path.read_text(encoding="utf-8"))
    pages = plan.get("pages") or []
    if args.only_page:
        pages = [p for p in pages if int(p["page_no"]) == args.only_page]
        if not pages:
            print(f"[book-render] page {args.only_page} not in plan", file=sys.stderr)
            return 2

    fal_key = (run_dir.parent.parent / ".config" / "fal.key").read_text(encoding="utf-8").strip()
    logger = VPSLogger(run_id=rs.run_id) if hasattr(VPSLogger, "__init__") else None
    fal = FalClient(key=fal_key, logger=logger, run_id=rs.run_id)
    analyzer = VisualAnalyzer(fal_client=fal, logger=logger, run_id=rs.run_id)

    plugin_root = Path(__file__).resolve().parent.parent
    templates_md = load_templates_md(plugin_root)

    char_brief_path = run_dir / "characters" / "description_block.md"
    character_brief = char_brief_path.read_text(encoding="utf-8") if char_brief_path.exists() else ""

    results = []
    for page in pages:
        try:
            qa = render_one_page(page, run_dir, fal, analyzer, templates_md, character_brief)
            results.append((int(page["page_no"]), qa.get("status", "unknown")))
        except Exception as e:  # noqa: BLE001
            print(f"[book-render] page {page['page_no']} failed: {e}", file=sys.stderr)
            results.append((int(page["page_no"]), "errored"))

    ok = sum(1 for _, s in results if s == "ok")
    print(f"[book-render] rendered {ok}/{len(results)} pages.")
    rs.phases.setdefault("book_render", {})["status"] = "complete" if ok == len(results) else "partial"
    rs.next_phase = "book_print_prep"
    rs.save()
    return 0 if ok == len(results) else 1


if __name__ == "__main__":
    raise SystemExit(main())
```

intended book-voice copy.

No external calls. Deterministic. Fast.

import argparse
import json
import sys
from pathlib import Path

from PIL import Image

from scripts.lib.book_canvas import compose
from scripts.lib.run_state import RunState


def main() -> int:
    ap = argparse.ArgumentParser()
